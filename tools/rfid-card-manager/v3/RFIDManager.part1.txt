# RFID Localhost Manager v3.0
# Dependency-free Windows companion for Arduino UNO R4 RFID reader.
# Uses only Windows PowerShell/.NET: no Python, no pip, no Arduino Wi-Fi.

$ErrorActionPreference = 'Stop'

$Script:Version = '3.0'
$Script:HostAddress = '127.0.0.1'
$Script:Port = 8765
$Script:Url = "http://localhost:$($Script:Port)/"
$Script:Baud = 115200
$Script:InstallRoot = Join-Path $env:LOCALAPPDATA 'Dallmayr\RFIDCardManager'
$Script:DataFile = Join-Path $Script:InstallRoot 'rfid_scans.json'
$Script:LogFile = Join-Path $Script:InstallRoot 'manager.log'
$Script:Scans = @()
$Script:Serial = $null
$Script:SerialPortName = ''
$Script:SerialMessage = 'Waiting for Arduino UNO R4...'
$Script:LastPortProbe = [DateTime]::MinValue
$Script:LastHostReply = [DateTime]::MinValue
$Script:LastUiSeen = [DateTime]::MinValue
$Script:LastPing = [DateTime]::MinValue
$Script:BrowserOpened = $false
$Script:Listener = $null

New-Item -ItemType Directory -Force -Path $Script:InstallRoot | Out-Null

function Write-Log([string]$Text) {
    try {
        $line = "{0} {1}" -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $Text
        Add-Content -LiteralPath $Script:LogFile -Value $line -Encoding UTF8
    } catch {}
}

function Load-Scans {
    if (-not (Test-Path -LiteralPath $Script:DataFile)) { return }
    try {
        $raw = Get-Content -LiteralPath $Script:DataFile -Raw -Encoding UTF8
        if ($raw.Trim()) {
            $data = $raw | ConvertFrom-Json
            if ($null -ne $data) { $Script:Scans = @($data) }
        }
    } catch {
        Write-Log "Could not load scan history: $($_.Exception.Message)"
        $Script:Scans = @()
    }
}

function Save-Scans {
    try {
        $json = ConvertTo-Json -InputObject @($Script:Scans) -Depth 5
        [IO.File]::WriteAllText($Script:DataFile, $json, [Text.UTF8Encoding]::new($false))
    } catch {
        Write-Log "Could not save scan history: $($_.Exception.Message)"
    }
}

function Escape-Html([string]$Text) {
    return [System.Net.WebUtility]::HtmlEncode([string]$Text)
}

function Escape-Xml([string]$Text) {
    if ($null -eq $Text) { return '' }
    return [System.Security.SecurityElement]::Escape([string]$Text)
}
