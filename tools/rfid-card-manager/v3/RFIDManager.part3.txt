            $Script:LastPing = Get-Date
        }
    } catch {
        Write-Log "Serial disconnected: $($_.Exception.Message)"
        $Script:SerialMessage = 'Reader disconnected; reconnecting...'
        Close-Serial
    }
}

function Get-StateJson {
    $copy = @($Script:Scans)
    $latest = if ($copy.Count -gt 0) { [string]$copy[-1].reverse_dec } else { '' }
    $unique = @($copy | ForEach-Object { $_.reverse_dec } | Where-Object { $_ } | Select-Object -Unique).Count
    $reversed = @($copy | Sort-Object number -Descending)
    $state = [ordered]@{
        serial = [ordered]@{
            connected = ($null -ne $Script:Serial -and $Script:Serial.IsOpen)
            port = $Script:SerialPortName
            message = $Script:SerialMessage
        }
        total = $copy.Count
        unique = $unique
        latest = $latest
        scans = $reversed
        version = $Script:Version
    }
    return ($state | ConvertTo-Json -Depth 7 -Compress)
}

function Get-CsvBytes {
    $sb = [Text.StringBuilder]::new()
    [void]$sb.AppendLine('Number,Scanned,UID,ID (dec),Reverse DEC,Card type')
    foreach ($x in @($Script:Scans)) {
        $vals = @($x.number,$x.timestamp,$x.uid,$x.id_dec,$x.reverse_dec,$x.card_type) | ForEach-Object {
            '"' + ([string]$_).Replace('"','""') + '"'
        }
        [void]$sb.AppendLine(($vals -join ','))
    }
    $utf8Bom = [byte[]](0xEF,0xBB,0xBF)
    $body = [Text.Encoding]::UTF8.GetBytes($sb.ToString())
    $all = New-Object byte[] ($utf8Bom.Length + $body.Length)
    [Array]::Copy($utf8Bom,0,$all,0,$utf8Bom.Length)
    [Array]::Copy($body,0,$all,$utf8Bom.Length,$body.Length)
    return $all
}

function Write-Utf8NoBom([string]$Path, [string]$Text) {
    $parent = Split-Path -Parent $Path
    if ($parent) { New-Item -ItemType Directory -Force -Path $parent | Out-Null }
    [IO.File]::WriteAllText($Path, $Text, [Text.UTF8Encoding]::new($false))
}
