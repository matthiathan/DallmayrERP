function Get-ArduinoCandidates {
    $result = @()
    try {
        $entities = Get-CimInstance Win32_PnPEntity -ErrorAction Stop | Where-Object {
            $_.Name -match '\(COM\d+\)' -and (
                $_.Name -match 'Arduino|UNO R4' -or
                $_.PNPDeviceID -match 'VID_2341|VID_2A03'
            )
        }
        foreach ($e in $entities) {
            if ($e.Name -match '\((COM\d+)\)') { $result += $matches[1] }
        }
    } catch {}

    if ($result.Count -eq 0) {
        try { $result = @([IO.Ports.SerialPort]::GetPortNames() | Sort-Object) } catch {}
    }
    return @($result | Select-Object -Unique)
}
